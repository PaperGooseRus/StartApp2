cmake_minimum_required(VERSION 3.20)
project(StartApp2 VERSION 0.7.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(PIXEL_BUILD_GAME "Build the SDL2 test game" ON)
option(PIXEL_BUILD_EDITOR "Build PixelStudio IDE" ${PIXEL_BUILD_GAME})
include(CTest)

add_library(pixel_engine STATIC src/world.cpp src/soil.cpp src/snapshot.cpp src/scene.cpp src/scene_io.cpp src/animation.cpp src/image.cpp)
target_include_directories(pixel_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(pixel_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(pixel_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(pixel_project STATIC game/project.cpp game/storage.cpp)
target_include_directories(pixel_project PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/game")
target_link_libraries(pixel_project PUBLIC pixel_engine)
if(WIN32)
    target_link_libraries(pixel_project PUBLIC shell32)
endif()
add_library(pixel_document STATIC editor/document.cpp)
target_include_directories(pixel_document PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/editor")
target_link_libraries(pixel_document PUBLIC pixel_project)
add_executable(pixel_project_demo examples/project.cpp)
target_link_libraries(pixel_project_demo PRIVATE pixel_project)
add_executable(pixel_demo examples/demo.cpp)
target_link_libraries(pixel_demo PRIVATE pixel_engine)
if(BUILD_TESTING)
    add_executable(pixel_tests tests/world_test.cpp)
    target_link_libraries(pixel_tests PRIVATE pixel_engine)
    add_test(NAME engine COMMAND pixel_tests)
    add_executable(soil_tests tests/soil_test.cpp)
    target_link_libraries(soil_tests PRIVATE pixel_engine)
    add_test(NAME soil COMMAND soil_tests)
    add_executable(scene_tests tests/scene_test.cpp)
    target_link_libraries(scene_tests PRIVATE pixel_engine)
    target_include_directories(scene_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/game")
    add_test(NAME scene COMMAND scene_tests)
    add_executable(snapshot_tests tests/snapshot_test.cpp game/storage.cpp)
    target_link_libraries(snapshot_tests PRIVATE pixel_engine)
    target_include_directories(snapshot_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/game")
    add_test(NAME snapshot COMMAND snapshot_tests)
    add_executable(studio_tests tests/studio_test.cpp)
    target_link_libraries(studio_tests PRIVATE pixel_document)
    add_test(NAME studio_model COMMAND studio_tests)
endif()

if(PIXEL_BUILD_GAME OR PIXEL_BUILD_EDITOR)
    # Use ordinary main() on all platforms, with SDL_SetMainReady().
    set(SDL2_NO_MWINDOWS ON)
    find_package(SDL2 CONFIG QUIET)
    if(NOT TARGET SDL2::SDL2)
        if(WIN32)
            message(FATAL_ERROR "SDL2 CMake config missing. Install the MSYS2 SDL2 package or set SDL2_DIR to the official SDK cmake directory.")
        endif()
        find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
        find_library(SDL2_LIBRARY NAMES SDL2)
        if(NOT SDL2_INCLUDE_DIR OR NOT SDL2_LIBRARY)
            message(FATAL_ERROR "SDL2 development files missing. See README.md for Linux/Windows installation, or use -DPIXEL_BUILD_GAME=OFF for the engine only.")
        endif()
        add_library(SDL2::SDL2 UNKNOWN IMPORTED)
        set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${SDL2_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${SDL2_INCLUDE_DIR}")
    endif()
    add_library(pixel_render STATIC game/render.cpp game/sprites.cpp)
    target_link_libraries(pixel_render PUBLIC pixel_project SDL2::SDL2)
    set(_pixel_apps)
    if(PIXEL_BUILD_GAME)
        add_executable(PixelSandbox game/main.cpp)
        target_link_libraries(PixelSandbox PRIVATE pixel_render)
        list(APPEND _pixel_apps PixelSandbox)
        if(BUILD_TESTING)
            add_test(NAME game_smoke COMMAND PixelSandbox --smoke-test)
            set_tests_properties(game_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        endif()
    endif()
    if(PIXEL_BUILD_EDITOR)
        add_library(pixel_imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
            third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
            third_party/imgui/backends/imgui_impl_sdl2.cpp third_party/imgui/backends/imgui_impl_sdlrenderer2.cpp)
        target_include_directories(pixel_imgui PUBLIC third_party/imgui third_party/imgui/backends)
        target_link_libraries(pixel_imgui PUBLIC SDL2::SDL2)
        add_executable(PixelEditor editor/main.cpp editor/application.cpp)
        target_link_libraries(PixelEditor PRIVATE pixel_document pixel_render pixel_imgui)
        list(APPEND _pixel_apps PixelEditor)
        install(FILES third_party/imgui/LICENSE.txt DESTINATION licenses RENAME IMGUI-LICENSE.txt)
        if(BUILD_TESTING)
            add_test(NAME editor_smoke COMMAND PixelEditor --smoke-test)
            set_tests_properties(editor_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        endif()
    endif()
    foreach(_app IN LISTS _pixel_apps)
        add_custom_command(TARGET ${_app} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:${_app}>/assets")
        if(WIN32)
            add_custom_command(TARGET ${_app} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E
                copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:${_app}>")
        endif()
        install(TARGETS ${_app} RUNTIME DESTINATION .)
    endforeach()
    add_custom_target(pixel_assets ALL COMMAND "${CMAKE_COMMAND}" -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/assets")
    foreach(_app IN LISTS _pixel_apps)
        add_dependencies(${_app} pixel_assets)
    endforeach()
    install(DIRECTORY assets DESTINATION .)
    install(FILES third_party/STB-LICENSE.txt DESTINATION licenses)
    install(FILES README.md DESTINATION .)
    install(FILES examples/example.pxproj DESTINATION examples)
    install(FILES docs/player-plan.md docs/layers.md docs/interactions.md docs/roadmap.md docs/preview-v0.6.png docs/editor.md docs/editor-level.png docs/editor-sprites.png DESTINATION docs)
    if(WIN32)
        install(FILES "$<TARGET_FILE:SDL2::SDL2>" DESTINATION .)
        if(PIXEL_BUILD_GAME)
            install(FILES packaging/start-windows.bat DESTINATION . RENAME StartGame.bat)
        endif()
        if(PIXEL_BUILD_EDITOR)
            install(FILES packaging/start-editor.bat DESTINATION . RENAME StartEditor.bat)
        endif()
        if(BUILD_TESTING)
            install(TARGETS pixel_tests soil_tests scene_tests snapshot_tests studio_tests RUNTIME DESTINATION .)
            install(FILES packaging/check-windows.bat DESTINATION . RENAME CheckGame.bat)
        endif()
        if(MINGW)
            install(FILES packaging/GCC-COPYING.txt packaging/GCC-RUNTIME-EXCEPTION.txt DESTINATION licenses)
            get_filename_component(_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
            foreach(_runtime libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
                if(EXISTS "${_compiler_dir}/${_runtime}")
                    install(FILES "${_compiler_dir}/${_runtime}" DESTINATION .)
                endif()
            endforeach()
        endif()
        install(FILES packaging/WINDOWS.txt DESTINATION . RENAME START-HERE.txt)
    elseif(UNIX AND NOT APPLE)
        if(PIXEL_BUILD_GAME)
            install(PROGRAMS packaging/start-linux.sh DESTINATION . RENAME StartGame.sh)
        endif()
        if(PIXEL_BUILD_EDITOR)
            install(PROGRAMS packaging/start-editor.sh DESTINATION . RENAME StartEditor.sh)
        endif()
        install(FILES packaging/LINUX.txt DESTINATION . RENAME START-HERE.txt)
    endif()
    # Redistribution notice for SDL2 when a DLL is included in a Windows package.
    install(FILES packaging/SDL2-LICENSE.txt packaging/LLVM-LICENSE.txt
        packaging/MINGW-LICENSE.txt packaging/THIRD-PARTY.txt DESTINATION licenses)
endif()
