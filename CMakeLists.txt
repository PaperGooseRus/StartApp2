cmake_minimum_required(VERSION 3.20)
project(StartApp2 VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(PIXEL_BUILD_GAME "Build the SDL2 test game" ON)
include(CTest)

add_library(pixel_engine STATIC src/world.cpp)
target_include_directories(pixel_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(pixel_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(pixel_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(pixel_demo examples/demo.cpp)
target_link_libraries(pixel_demo PRIVATE pixel_engine)
if(BUILD_TESTING)
    add_executable(pixel_tests tests/world_test.cpp)
    target_link_libraries(pixel_tests PRIVATE pixel_engine)
    add_test(NAME engine COMMAND pixel_tests)
endif()

if(PIXEL_BUILD_GAME)
    # Use ordinary main() on all platforms, with SDL_SetMainReady().
    set(SDL2_NO_MWINDOWS ON)
    find_package(SDL2 CONFIG QUIET)
    if(NOT TARGET SDL2::SDL2)
        if(WIN32)
            message(FATAL_ERROR "SDL2 CMake config missing. Install the MSYS2 SDL2 package or set SDL2_DIR to the official SDK cmake directory.")
        endif()
        find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
        find_library(SDL2_LIBRARY NAMES SDL2)
        if(NOT SDL2_INCLUDE_DIR OR NOT SDL2_LIBRARY)
            message(FATAL_ERROR "SDL2 development files missing. See README.md for Linux/Windows installation, or use -DPIXEL_BUILD_GAME=OFF for the engine only.")
        endif()
        add_library(SDL2::SDL2 UNKNOWN IMPORTED)
        set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${SDL2_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${SDL2_INCLUDE_DIR}")
    endif()
    add_executable(PixelSandbox game/main.cpp)
    target_link_libraries(PixelSandbox PRIVATE pixel_engine SDL2::SDL2)
    if(BUILD_TESTING)
        add_test(NAME game_smoke COMMAND PixelSandbox --smoke-test)
        set_tests_properties(game_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    endif()
    install(TARGETS PixelSandbox RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
    if(WIN32)
        # Copy SDL2.dll beside the executable for both build-tree and packaged launches.
        add_custom_command(TARGET PixelSandbox POST_BUILD COMMAND "${CMAKE_COMMAND}" -E
            copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:PixelSandbox>")
        install(FILES "$<TARGET_FILE:SDL2::SDL2>" DESTINATION .)
        install(FILES packaging/start-windows.bat DESTINATION . RENAME StartGame.bat)
        if(MINGW)
            install(FILES packaging/GCC-COPYING.txt packaging/GCC-RUNTIME-EXCEPTION.txt DESTINATION licenses)
            get_filename_component(_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
            foreach(_runtime libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
                if(EXISTS "${_compiler_dir}/${_runtime}")
                    install(FILES "${_compiler_dir}/${_runtime}" DESTINATION .)
                endif()
            endforeach()
        endif()
        install(FILES packaging/WINDOWS.txt DESTINATION . RENAME START-HERE.txt)
    elseif(UNIX AND NOT APPLE)
        install(PROGRAMS packaging/start-linux.sh DESTINATION . RENAME StartGame.sh)
        install(FILES packaging/LINUX.txt DESTINATION . RENAME START-HERE.txt)
    endif()
    # Redistribution notice for SDL2 when a DLL is included in a Windows package.
    install(FILES packaging/SDL2-LICENSE.txt packaging/LLVM-LICENSE.txt
        packaging/MINGW-LICENSE.txt packaging/THIRD-PARTY.txt DESTINATION licenses)
endif()
